// ── ENTRY POINTS ─────────────────────────────────────── function doPost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var raw = ""; // Read from form field, postData, or querystring if (e.parameter && e.parameter.payload) { raw = e.parameter.payload; } else if (e.postData && e.postData.contents) { raw = e.postData.contents; } else if (e.parameter && e.parameter.data) { raw = e.parameter.data; } if (!raw) return ok_({ok:false, error:"No data received"}); var data = JSON.parse(raw); var action = data.action || "saveDespacho"; // Log for debugging Logger.log("action: " + action); if (data.bultos) Logger.log("bultos count: " + data.bultos.length); if (action === "saveClientes") return ok_(guardarClientes_(ss, data.clientes)); if (action === "saveCliente") return ok_(guardarCliente_(ss, data.cliente)); if (action === "deleteCliente") return ok_(eliminarCliente_(ss, data.id)); if (action === "getClientes") return ok_(getClientes_(ss)); return ok_(guardarDespacho_(ss, data)); } catch(err) { Logger.log("doPost error: " + err.message); return ok_({ok:false, error:err.message}); } } function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var act = (e.parameter && e.parameter.action) || "getClientes"; if (act === "getClientes") return ok_(getClientes_(ss)); if (act === "getDespachos") return ok_(getDespachos_(ss)); return ok_({ok:true, ping:true}); } catch(err) { return ok_({ok:false, error:err.message}); } } // ── HELPERS ──────────────────────────────────────────── function ok_(data) { var output = ContentService.createTextOutput( typeof data === "string" ? data : JSON.stringify(data) ); output.setMimeType(ContentService.MimeType.JSON); return output; } function sheet_(ss, name, headers, color) { var h = ss.getSheetByName(name); if (!h) { h = ss.insertSheet(name); } // Always ensure headers match (update if column count differs) var lastCol = h.getLastColumn(); if (lastCol !== headers.length || h.getLastRow() === 0) { // Clear row 1 and rewrite headers if (h.getLastRow() > 0) { h.getRange(1,1,1,Math.max(lastCol,headers.length)).clearContent(); } h.getRange(1,1,1,headers.length).setValues([headers]); h.getRange(1,1,1,headers.length) .setFontWeight("bold").setBackground(color).setFontColor("white"); h.setFrozenRows(1); } return h; } // ── DESPACHOS ────────────────────────────────────────── function guardarDespacho_(ss, d) { var h = sheet_(ss,"Despachos", ["ID","Fecha","Tipo","Cliente","Guia","OC","OV","Picking", "Ciudad","Total Bultos","TipoBulto","Ruta", "N° Bulto","Tipo Bulto","Alto","Ancho","Largo","Medidas","Peso","Timestamp"], "#D0191A"); var bultos=(d.bultos&&d.bultos.length)?d.bultos:[{num:1,alto:"",ancho:"",largo:"",dims:"",peso:""}]; bultos.forEach(function(b){ h.appendRow([ d.id||"", d.fecha||"", d.tipo||"", d.destinatario||"", d.gd||d.guia||"", d.oc||"", d.ov||"", d.picking||"", d.ciudad||"", d.total||1, d.tipoBulto||"", d.ruta||"", b.num||1, b.tipo||"", b.alto||"", b.ancho||"", b.largo||"", b.dims||"", b.peso||"", new Date().toLocaleString("es-CL") ]); }); // ── Duplicar en hoja VENTA WEB si tipo='ventaweb' ──────────────────── if ((d.tipo||"").toLowerCase() === "ventaweb") { var hVW = sheet_(ss,"VENTA WEB", ["ID","Fecha","Nombre","RUT","Dirección","Teléfono","Email", "Guia","OC","OV","Picking","Total Bultos", "N° Bulto","Tipo Bulto","Alto","Ancho","Largo","Medidas","Peso","Timestamp"], "#7c3aed"); }); }); return {ok:true, count:clientes.length}; } function eliminarCliente_(ss, id) { var h = ss.getSheetByName("Clientes"); if (h) borrar_(h, id); return {ok:true}; } function borrar_(h, id) { for (var i=h.getLastRow(); i>=2; i--) if (String(h.getRange(i,1).getValue()).trim()===String(id).trim()) h.deleteRow(i); }
